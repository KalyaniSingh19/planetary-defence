Measures used

Total Approaches = COUNTROWS(Fact_approaches)
Unique Asteroids = DISTINCTCOUNT(Fact_approaches[ObjectID])
Closest Approach LD = MIN(Fact_approaches[Distance_LD])
Average Velocity = AVERAGE(Fact_approaches[Velocity_kms])
PHA Approaches = CALCULATE([Total Approaches], Dim_obj[pha] = "Y")
